# 统一作品详情组件 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把 ASMR ONE / DLsite / 作品库三个作品详情页合并为一个共享组件（展示外壳 + 逻辑 composable + 三个薄路由），并新增按 RJ 查本地作品的接口作为本地状态单一来源。

**Architecture:** 三份逐字重复的模板收敛为纯展示组件 `WorkDetailView.vue`（零请求、零 `source` 分支）；全部数据获取与按钮决策收敛到 `useWorkDetail.ts`；模式判定与按钮矩阵抽为纯函数 `src/lib/workDetail.ts`（可用 Node 内置 `node:test` 做 TDD）；三个路由文件退化为薄壳。后端新增 `GET /api/works/by-rj/{rj}` 补齐 DLsite 页缺失的本地状态。

**Tech Stack:** Rust + axum 0.8 + rusqlite/r2d2（后端）；Vue 3 `<script setup>` + TS + vue-router hash 模式 + Pinia + Tailwind + lucide-vue-next（前端）；`node --test`（Node 26 原生跑 TS，无新依赖）。

**Spec:** `docs/superpowers/specs/2026-09-16-统一作品详情组件-design.md`

## Global Constraints

- **不改动** `TrackList.vue` 的 props 契约（只用既有 `online` / `tree` 开关）、`AsmrFileTree.vue` 内部实现、`AsmrOnlineWork` / `ScrapedWork` 结构体、下载引擎、`/api/download/asmr`。
- **不改动** 作品列表页、卡片组件、播放器页。
- **不新增前端测试框架**：只用 Node 内置 `node:test`（已实测 Node v26.8.2 可原生运行 `.ts` 测试，无需额外依赖）。
- **Node 原生 TS 限制（实测结论，务必遵守）**：Node 只做类型剥离，不做路径解析。因此：
  - 测试文件与被测模块之间的相对导入**必须写 `.ts` 扩展名**（`tsconfig.json` 已开 `allowImportingTsExtensions: true`）。
  - `src/lib/workDetail.ts` 内**不得使用省略扩展名的相对「值」导入**（`import type` 会被整体剥离，写不存在的模块也不会报错，属例外）。`workDetail.ts` 只允许 `import type { … } from "./types"` 这一种相对导入。
- 按钮文案：`ASMR One下载`（`source === 'work'` 时）/ `下载`（ASMR ONE、DLsite 页）；`在线播放`（原「在线试听」改名）。
- 移除「本地文件夹」按钮及 `openLocal()`（决策 12）。
- 后端命令一律用 `cargo test --manifest-path server/Cargo.toml`（仓库无 workspace 根 Cargo.toml）。
- 前端类型检查一律用 `npm run build`（= `vue-tsc --noEmit && vite build`）。
- 所有提交信息用中文，遵循仓库既有 `feat: …` / `fix: …` / `docs: …` 前缀风格。
- 开发服务已在跑：前端 `http://localhost:1420`，后端 `http://127.0.0.1:1421`；shell 有 `http_proxy`，curl 必须加 `--noproxy '*'`；Vite 只监听 IPv6，用 `localhost` 而非 `127.0.0.1`。
- 重启后端（Rust 改动后必须）：`pkill -f "target/debug/asmr-db" && ASMR_DB_DIR="$PWD/data" cargo run --manifest-path server/Cargo.toml`

在 `server/src/handlers/works.rs` 的 `get_work` 函数之后插入：

```rust
/// 按 RJ 号查本地作品（大小写不敏感）。未入库返回 `null`。
/// DLsite / ASMR ONE 详情页用它获取本地状态（下载状态、音轨数、标签、分组）。
pub async fn get_work_by_rj(
    State(state): State<SharedState>,
    Path(rj): Path<String>,
) -> Result<Json<Option<WorkView>>, AppError> {
    let view = tokio::task::spawn_blocking(move || -> Result<Option<WorkView>, AppError> {
        let conn = state.db.get().map_err(AppError::new)?;
        let work = queries::get_work_by_rj(&conn, &rj).map_err(AppError::new)?;
        match work {
            Some(w) => Ok(Some(queries::work_view(&conn, w).map_err(AppError::new)?)),
            None => Ok(None),
        }
    })
    .await
    .map_err(AppError::new)??;
    Ok(Json(view))
}
```

```ts
 * 作品详情页的纯逻辑：模式判定、按钮矩阵、元数据行、状态角标、三种来源归一化。
 *
 * 本文件被 `node --test` 直接加载（Node 原生类型剥离），因此：
 * 除了下面这一行 `import type`，不得出现任何相对「值」导入

export type DetailMode = "local" | "asmr" | "dlsite";

export type ActionKey =
  | "play-all"
  | "online-play"
  | "import"
  | "go-local"
  | "asmr-download"
  | "dl-download"
  | "scan-tracks"
  | "bind-local"

  | "delete"
  | "open-dlsite"
  | "open-asmr";

export type ActionVariant = "primary" | "outline" | "ghost";

  text: string;
}

export interface StatusBadge {

  tone: "green" | "accent" | "neutral" | "yellow";

}

export interface DetailAction {

  label: string;
  variant: ActionVariant;
}

export interface ModeInput {
  source: DetailSource;
  hasOnline: boolean;
  localTrackCount: number;

  source: DetailSource;
  mode: DetailMode;
  hasOnline: boolean;

/** spec §2.2 模式判定公式 */
export function resolveMode(input: ModeInput): DetailMode {
  const { source, hasOnline, localTrackCount } = input;
  if (source === "asmr") return "asmr";
  if (source === "work" && localTrackCount > 0) return "local";
  return hasOnline ? "asmr" : "dlsite";
}

/** spec §4 按钮矩阵，返回按钮键的有序列表 */
export function resolveActions(input: ActionInput): ActionKey[] {
  const { source, mode, hasOnline, hasLocalWork, hasRj } = input;
  const keys: ActionKey[] = [];

  if (source === "work") {
    // 作品库：本地操作在前，在线操作在后（场景 4/4b 顺序）
    if (mode === "local") keys.push("play-all");
    if (mode === "local") keys.push("dl-download");
    if (mode !== "local") {
      if (hasOnline) keys.push("online-play", "asmr-download");
      keys.push("dl-download");
    }
    keys.push("scan-tracks", "bind-local", "translate");
    if (mode === "local" && hasOnline) keys.push("online-play", "asmr-download");
  } else {
    // ASMR ONE / DLsite：先在线播放，再入库或本地详情，再 asmr.one 下载
    if (hasOnline) keys.push("online-play");
    if (hasLocalWork) keys.push("go-local");
    else if (hasRj) keys.push("import");
    if (hasOnline) keys.push("asmr-download");
  }

  if (hasRj) keys.push("open-dlsite");
  // spec §4.1：仅 DLsite 页的 ASMR One 外链受 hasOnline 约束
  if (source === "dlsite" ? hasOnline : true) keys.push("open-asmr");

  if (source === "work") keys.push("delete");

  return keys;
}

/** 按钮文案（spec §4.2） */

  switch (key) {

      return "在线播放";
    case "import":
      return "入库";

      return "下载";

      return "扫描音轨";

      return "删除";
    case "open-dlsite":
      return source === "work" ? "DLsite 页面" : "DLsite";

}

/** 按钮样式：首个为 primary；外链/关联本地/删除为 ghost；其余 outline */
export function actionVariant(key: ActionKey, index: number): ActionVariant {
  if (index === 0) return "primary";
  if (key === "open-dlsite" || key === "open-asmr" || key === "bind-local" || key === "delete") {
    return "ghost";
  }
  return "outline";
}

/** 元数据行（spec §4 三个页面的元数据逐字一致，仅 rating 为 ASMR ONE 独有） */
export function buildMeta(
  _source: DetailSource,
  w: Work,
  rating?: number | null,
  rateCount?: number | null
): MetaItem[] {
  const items: MetaItem[] = [];
  if (w.circle_name) items.push({ icon: "🏢", text: `社团：${w.circle_name}` });
  if (w.sale_date) items.push({ icon: "📅", text: `发售：${String(w.sale_date).slice(0, 10)}` });
  if (rating != null && rating > 0) {
    items.push({
      icon: "⭐",
      text: rateCount ? `${rating.toFixed(1)}（${rateCount} 人评价）` : rating.toFixed(1),
    });
  }
  if (w.price) items.push({ icon: "💰", text: `${w.price.toLocaleString()}円` });
  if (w.duration_min) items.push({ icon: "⏱", text: `${w.duration_min}分` });
  if (w.file_size_mb) items.push({ icon: "📦", text: `${w.file_size_mb.toFixed(1)}MB` });
  if (w.age_class) items.push({ icon: "", text: w.age_class.toUpperCase() });
  return items;
}

/**
 * 状态角标（spec §7）。
 * 已入库 → 显示本地下载状态；未入库 → 一律「未入库」（黄）。
 * `clickable` 由调用方给出（仅「非作品库页 + 已入库」为真），未入库恒不可点击。
 */
export function resolveStatusBadge(
  downloadStatus: string,
  hasLocalWork: boolean,
  clickable: boolean
): StatusBadge {
  if (!hasLocalWork) {
    return { text: "未入库", tone: "yellow", clickable: false };
  }
  const canClick = clickable;
  if (downloadStatus === "downloaded") {
    return { text: "已下载", tone: "green", clickable: canClick };
  }
  if (downloadStatus === "downloading") {
    return { text: "下载中", tone: "accent", clickable: canClick };
  }
  return { text: "未下载", tone: "neutral", clickable: canClick };
}

/**
 * 归一化：本地 WorkView 优先；本地缺失时用在线/抓取数据构造临时 WorkView。
 * 这是原先 ASMR ONE 与 DLsite 两处重复 `fakeView` 的唯一实现处。
 */
export function buildWorkView(
  work: Work,
  tags: Tag[],
  actors: string[],
  local: WorkView | null,
  trackCount: number,
  totalDurationSec: number
): WorkView {
  return {
    work,
    tags: local ? local.tags : tags,
    actors: local ? local.actors : actors,
    download_status: local ? local.download_status : "not_downloaded",
    local_path: local ? local.local_path : null,
    track_count: trackCount,
    total_duration_sec: totalDurationSec,
    group: local ? local.group : null,
  };
}

/**
 * 把 ASMR ONE / DLsite 的标签名数组转成只读 Tag 形态，供统一展示组件渲染。
 * 负数 id 表示「非本地标签」，仅用于 v-for key，不会被提交回后端。
 */
export function tagsFromNames(names: string[]): Tag[] {
  return names.map((name, i) => ({
    id: -(i + 1),
    name,
    color: "#888888",
    is_preset: false,
  }));
}

/** 由 ASMR ONE / DLsite 数据构造临时 Work（id 为 0 表示未入库） */
export function buildFakeWork(

  a: AsmrOnlineWork | null,

  const rj = a?.rj_code ?? s?.rj_code ?? String(id);

    sale_date: a?.release ?? s?.sale_date ?? null,
    description: a?.description ?? s?.description ?? null,
    description_zh: source === "asmr" ? (a?.description ?? null) : null,
    age_class: s?.age_class ?? null,
    duration_min: a?.duration_sec ? Math.round(a.duration_sec / 60) : (s?.duration_min ?? null),
    file_size_mb: s?.file_size_mb ?? null,
    dlsite_url: s?.dlsite_url ?? `https://www.dlsite.com/maniax/work/=/product_id/${rj}.html`,
    created_at: null,
    updated_at: null,
  };
}
```

```ts
import { useRouter } from "vue-router";
import * as api from "../lib/api";
import type { AsmrDownloadFile, AsmrOnlineWork, AsmrTreeNode, ScrapedWork, Tag, Track, Work, WorkView } from "../lib/types";

export function useWorkDetail(opts: { source: DetailSource; id: Ref<string> }): UseWorkDetail {
  const router = useRouter();
  const player = usePlayerStore();
  const worksStore = useWorksStore();
  const downloadStore = useDownloadStore();

  const loading = ref(true);
  const error = ref("");
  const copied = ref(false);

  // 主数据（按 source 三选一）
  const localView = ref<WorkView | null>(null);
  const asmrWork = ref<AsmrOnlineWork | null>(null);
  const scrapedWork = ref<ScrapedWork | null>(null);
  const dlsiteDescription = ref("");
  const descriptionLoading = ref(false);

  const hasOnline = ref(false);
  const localTracks = ref<Track[]>([]);
  const onlineTracks = ref<Track[]>([]);
  const fileTree = ref<AsmrTreeNode[] | null>(null);
  const tracksLoading = ref(false);
  const treeLoading = ref(false);
  const busyKey = ref<ActionKey | null>(null);

  const showFileTree = ref(false);
  const showFolderPicker = ref(false);
  const showUrlInput = ref(false);
  const dlUrl = ref("");

  const rjCode = computed<string | null>(
    () => localView.value?.work.rj_code ?? asmrWork.value?.rj_code ?? scrapedWork.value?.rj_code ?? null
  );

  const hasLocalWork = computed(() => localView.value != null);

  /** spec §2.2 模式判定 */
  const mode = computed<DetailMode>(() =>

      source: opts.source,

      scrapedWork.value?.title_zh ||
      scrapedWork.value?.title_ja ||
      rjCode.value ||
      `#${opts.id.value}`
  );

  const subtitle = computed<string | null>(() => {
    const ja = rawWork.value?.title_ja ?? null;
    return ja && ja !== title.value ? ja : null;

  const coverUrl = computed<string | null>(() => rawWork.value?.cover_url ?? null);

  );

    () => localView.value?.actors ?? asmrWork.value?.vas ?? scrapedWork.value?.actors ?? []
  );

  /**
   * 标签：已入库时用本地标签（作品库单一来源，含用户编辑）；
   * 未入库时用来源自身的标签名（asmr.one / DLsite），转为只读 Tag 形态。

    if (localView.value) return localView.value.tags;
    if (opts.source === "asmr") return tagsFromNames(asmrWork.value?.tags ?? []);
    if (opts.source === "dlsite") return tagsFromNames(scrapedWork.value?.tags ?? []);

  const workId = computed<number | null>(() => localView.value?.work.id ?? null);

      return localView.value?.work.description_zh ?? localView.value?.work.description ?? null;
    }
    return asmrWork.value?.description ?? scrapedWork.value?.description ?? dlsiteDescription.value ?? null;
  });

  const descriptionNote = computed<string | null>(() => {
    if (opts.source === "work") {
      return localView.value?.work.description_zh ? "中文翻译" : null;
    }
    if (opts.source === "asmr") {
      return !asmrWork.value?.description && dlsiteDescription.value ? "（来自 DLsite）" : null;
    }
    return null;
  });

  /** spec §8：local → 本地音轨；asmr → 在线音轨；dlsite → 空 */
  const displayTracks = computed<Track[]>(() =>

  );

  const view = computed<WorkView | null>(() => {
    if (opts.source === "work" && localView.value) {
      if (mode.value === "local") return localView.value;
      return {
        ...localView.value,
        track_count: onlineTracks.value.length,
        total_duration_sec: onlineTracks.value.reduce((s, t) => s + (t.duration_sec || 0), 0),
      };

    const w = rawWork.value;
    if (!w) return null;
    return buildWorkView(
      w,
      tags.value,
      actors.value,
      localView.value,
      displayTracks.value.length,
      displayTracks.value.reduce((s, t) => s + (t.duration_sec || 0), 0)
    );
  });

  const actions = computed<DetailAction[]>(() => {
    const keys = resolveActions({
      source: opts.source,
      mode: mode.value,
      hasOnline: hasOnline.value,
      hasLocalWork: hasLocalWork.value,
      hasRj: rjCode.value != null,
    });
    return keys.map((key, i) => ({
      key,
      label: actionLabel(key, opts.source),
      variant: actionVariant(key, i),
    }));
  });

  const notFound = computed(
    () =>
      !loading.value &&
      !error.value &&
      localView.value == null &&
      asmrWork.value == null &&
      scrapedWork.value == null
  );

  /** 探测 asmr.one 是否存在。spec §2.3：始终探测（不拉文件树），无 token 直接跳过。 */
  async function probeOnline(): Promise<boolean> {
    const rj = rjCode.value;
    if (!rj) return false;
    try {
      const s = await api.getSettings();
      if (!s.asmr_one_token) return false;
    } catch {
      return false;
    }
    try {
      const tracks = await api.listAsmroneTracks(rj);
      onlineTracks.value = tracks;
      return tracks.length > 0;
    } catch {
      onlineTracks.value = [];
      return false;
    }
  }

  async function loadLocalState() {
    const rj = rjCode.value;
    if (!rj) {
      localView.value = null;
      return;
    }
    try {
      localView.value = await api.getWorkByRj(rj);
    } catch {
      localView.value = null;
    }
  }

  async function load() {
    loading.value = true;
    error.value = "";
    copied.value = false;
    localView.value = null;
    asmrWork.value = null;
    scrapedWork.value = null;
    dlsiteDescription.value = "";
    localTracks.value = [];
    onlineTracks.value = [];
    fileTree.value = null;

    const id = opts.id.value;
    try {
      if (opts.source === "work") {
        const v = await api.getWork(Number(id));
        if (!v) {
          localView.value = null;
          return;
        }
        localView.value = v;
        localTracks.value = await api.listTracks(v.work.id);
      } else if (opts.source === "asmr") {
        asmrWork.value = await api.asmrWorkDetail(id);
        loadDlsiteDescription();
      } else {
        scrapedWork.value = await api.previewByRj(id);
      }
      await loadLocalState();
      tracksLoading.value = true;
      hasOnline.value = await probeOnline();
      // asmr 模式需要文件树以支撑「全部文件」标签页（spec §8）。
      // 作品库本地有音轨时 mode === 'local'，按 spec §2.3 不拉树（避免「音频 N / 全部文件 M」来源不一致）。
      if (mode.value === "asmr" && rjCode.value) {
        fileTree.value = await api.listAsmroneTree(rjCode.value).catch(() => null);
      }
      tracksLoading.value = false;
    } catch (e) {
      error.value = String(e);
    } finally {
      loading.value = false;
    }
  }

  /** ASMR ONE 页补充 DLsite 简介（asmr.one API 不提供简介） */
  async function loadDlsiteDescription() {
    const rj = asmrWork.value?.rj_code;
    if (!rj || descriptionLoading.value) return;
    descriptionLoading.value = true;
    try {
      dlsiteDescription.value = await api.getDlsiteDescription(rj);
    } catch {
      dlsiteDescription.value = "";
    } finally {
      descriptionLoading.value = false;
    }
  }

  function onTagsUpdated() {
    load();
  }

  function copiedReset() {
    copied.value = true;
    setTimeout(() => (copied.value = false), 1500);
  }

  async function playAll() {
    const w = view.value;
    if (!w || !localTracks.value.length) return;
    const progress = await api.getPlayProgress(w.work.id);
    let startIndex = 0;
    let initialTime = 0;
    if (progress) {
      const idx = localTracks.value.findIndex((t) => t.id === progress.track_id);
      if (idx >= 0) {
        startIndex = idx;
        initialTime = progress.last_position || 0;
      }
    }
    player.playFromWork(w, localTracks.value, startIndex, initialTime);
    router.push({ name: "player" });
  }

  async function playOnline() {
    const w = view.value;
    if (!w) return;
    if (!onlineTracks.value.length) {
      alert("asmr.one 未找到音频文件");
      return;
    }
    player.playFromWork(w, onlineTracks.value, 0, 0);
    router.push({ name: "player" });
  }

  async function importWork() {
    const rj = rjCode.value;
    if (!rj) return;
    const res = await api.importByRj(rj);
    worksStore.fetchWorks();
    worksStore.loadTags();
    router.push(`/work/${res.work.work.id}`);
  }

  function goLocal() {
    if (localView.value) router.push(`/work/${localView.value.work.id}`);
  }

  /** spec §6 下载流程：静默入库 → 拉文件树 → 弹选择框 → 下载 → 留在当前页 */
  async function openAsmrDownload() {
    const rj = rjCode.value;
    if (!rj) return;
    const s = await api.getSettings();
    if (!s.asmr_one_token) {
      alert("请先在「设置 -> asmr.one」登录获取 Token 后再试");
      return;
    }
    if (!localView.value) {
      const res = await api.importByRj(rj);
      worksStore.fetchWorks();
      worksStore.loadTags();
      localView.value = res.work;
    }
    showFileTree.value = true;
    // asmr 模式下文件树已在 load() 中拉好，避免重复请求
    if (fileTree.value) return;
    treeLoading.value = true;
    try {
      fileTree.value = await api.listAsmroneTree(rj);
    } catch (e) {
      alert(`获取 asmr.one 文件树失败: ${e}`);
      showFileTree.value = false;
    } finally {
      treeLoading.value = false;
    }
  }

  async function onFileTreeDownload(files: AsmrDownloadFile[]) {
    showFileTree.value = false;
    const id = localView.value?.work.id;
    if (!id) return;
    try {
      await api.downloadAsmroneFiles(id, files);
      alert(`已开始从 asmr.one 下载 ${files.length} 个文件，可在「下载管理」查看实时进度。`);
      await loadLocalState();
    } catch (e) {
      alert(`下载失败: ${e}`);
    }
  }

  function openDlsite() {
    const url =
      rawWork.value?.dlsite_url ??
      (rjCode.value
        ? `https://www.dlsite.com/maniax/work/=/product_id/${rjCode.value}.html`
        : null);
    if (url) window.open(url, "_blank", "noopener");
  }

  function openAsmrOne() {
    const rj = rjCode.value;
    if (rj) window.open(`https://www.asmr.one/work/${rj}`, "_blank", "noopener");
  }

  /** 作品库「下载」：首次点击展开内联直链输入框，填好后再点才真正开始下载 */
  async function startDlDownload() {
    const id = localView.value?.work.id;
    if (!id) return;
    if (!dlUrl.value.trim()) {
      showUrlInput.value = true;
      return;
    }
    await downloadStore.startDownload(id, dlUrl.value.trim());
    showUrlInput.value = false;
  }

  async function scanTracks() {
    const id = localView.value?.work.id;
    if (!id) return;
    const res = await api.scanAudioTracks(id);
    if (!res.ok) alert(res.message || "扫描失败");
    else await load();
  }

  async function translateWork() {
    const id = localView.value?.work.id;
    if (!id) return;
    const res = await api.enqueueTranslation(id);
    alert(res.message || "已加入翻译队列");
  }

  async function deleteWork() {
    const id = localView.value?.work.id;
    if (!id) return;
    if (!confirm(`确定删除「${title.value}」？将从数据库移除该作品。`)) return;
    await api.deleteWork(id);
    worksStore.removeWork(id);
    router.push({ name: "home" });
  }

  async function onFolderPicked(dir: string) {
    showFolderPicker.value = false;
    const rj = rjCode.value;
    if (!rj) return;
    try {
      const res = await api.bindLocalFolder(rj, dir);
      alert(
        `关联成功！扫描到 ${res.trackCount} 首音轨${res.groupName ? `，已自动分配分组「${res.groupName}」` : ""}`
      );
      await load();
      worksStore.fetchWorks();
      worksStore.loadGroups();
    } catch (e) {
      alert(`关联失败: ${e}`);
    }
  }

  /** 动作分发：统一处理 loading 状态与错误提示 */
  async function runAction(key: ActionKey) {
    if (busyKey.value) return;
    busyKey.value = key;
    try {
      switch (key) {
        case "play-all":
          await playAll();
          break;
        case "online-play":
          await playOnline();
          break;
        case "import":
          await importWork();
          break;
        case "go-local":
          goLocal();
          break;
        case "asmr-download":
          await openAsmrDownload();
          break;
        case "dl-download":
          await startDlDownload();
          break;
        case "scan-tracks":
          await scanTracks();
          break;
        case "bind-local":
          showFolderPicker.value = true;
          break;
        case "translate":
          await translateWork();
          break;
        case "delete":
          await deleteWork();
          break;
        case "open-dlsite":
          openDlsite();
          break;
        case "open-asmr":
          openAsmrOne();
          break;
      }
    } catch (e) {
      alert(`操作失败: ${e}`);
    } finally {
      busyKey.value = null;
    }
  }

  watch(opts.id, (v, old) => {
    if (v && v !== old) load();
  });
  load();

  return {
    loading,
    error,
    mode,
    hasOnline,
    localView,
    view,
    title,
    subtitle,
    rjCode,
    coverUrl,
    statusBadge,
    meta,
    actors,
    tags,
    editableTags,
    workId,
    description,
    descriptionNote,
    descriptionLoading,
    actions,
    localTracks,
    onlineTracks,
    fileTree,
    displayTracks,
    tracksLoading,
    treeLoading,
    notFound,
    copied,
    showFileTree,
    showFolderPicker,
    showUrlInput,
    dlUrl,
    busyKey,
    copiedReset,
    reload: load,
    runAction,
    onFileTreeDownload,
    onFolderPicked,
    onTagsUpdated,
  };
}
```
